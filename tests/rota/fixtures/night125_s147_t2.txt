```python
def test_echo_json_compact():
    import sys
    from io import StringIO
    from src.click.utils import echo_json
    
    old_stdout = sys.stdout
    sys.stdout = StringIO()
    
    try:
        result = echo_json({"key": "value"})
        
        # If result is a file-like object (EncodedFile), read from it
        if hasattr(result, 'read'):
            output = result.read()
        else:
            # Fallback to stdout capture
            sys.stdout = old_stdout
            output = sys.stdout.getvalue()
    finally:
        sys.stdout = old_stdout
    
    assert output.strip() == '{"key": "value"}'
    # Check for single newline at end
    assert output.endswith("\n")
    # Check no extra newlines
    assert output.count("\n") == 1
```
